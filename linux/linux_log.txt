          chmod +x sdk.sh

        Cross-Compiling ZeroMQ
        
          cd libzmq
          mkdir build && cd build
          cmake .. \
          -DCMAKE_INSTALL_PREFIX=$SDKTARGETSYSROOT/usr \
          -DCMAKE_C_COMPILER=aarch64-xilinx-linux-gcc \
          -DCMAKE_CXX_COMPILER=aarch64-xilinx-linux-g++ \
          -DCMAKE_SYSROOT=$SDKTARGETSYSROOT \
          -DCMAKE_FIND_ROOT_PATH=$SDKTARGETSYSROOT \
          -DCMAKE_FIND_ROOT_PATH_MODE_PROGRAM=NEVER \
          -DCMAKE_FIND_ROOT_PATH_MODE_LIBRARY=ONLY \
          -DCMAKE_FIND_ROOT_PATH_MODE_INCLUDE=ONLY \
          -DWITH_PERF_TOOL=OFF
  
  
5-11-2025
---------

  Middleware     
    
    Here’s a step-by-step guide from scratch to cross-compile a ZeroMQ-based C++ app for t
    he Trenz TE0802 board running PetaLinux,   using the PetaLinux 2024.2 SDK. This includes building 
    libzmq, linking it to your app, and deploying the result.
    
    STEP 1: Prepare Folder Structure
      cd rover/software/
      mkdir -p libzmq/build
      mkdir -p libzmq/install
      mkdir -p zeromq_pub_project/build

    STEP 2: Source the PetaLinux SDK
      source /tools/Xilinx/Petalinux/2024.2/sdk/environment-setup-cortexa72-cortexa53-xilinx-linux 
    
    STEP 3: Clone and Build ZeroMQ
      Clone libzmq repo outside project folder
        git clone https://github.com/zeromq/libzmq.git src
      Copy and paste clone libzmq  folder in the followung repo
        software/libzmq/
      cd build
      cmake ../libzmq \
        -DCMAKE_INSTALL_PREFIX=$SDKTARGETSYSROOT/usr \
        -DCMAKE_C_COMPILER=aarch64-xilinx-linux-gcc \
        -DCMAKE_CXX_COMPILER=aarch64-xilinx-linux-g++ \
        -DCMAKE_SYSROOT=$SDKTARGETSYSROOT \
        -DCMAKE_FIND_ROOT_PATH=$SDKTARGETSYSROOT \
        -DCMAKE_FIND_ROOT_PATH_MODE_PROGRAM=NEVER \
        -DCMAKE_FIND_ROOT_PATH_MODE_LIBRARY=ONLY \
        -DCMAKE_FIND_ROOT_PATH_MODE_INCLUDE=ONLY \
        -DWITH_PERF_TOOL=OFF \
        -DWITH_LIBBSD=OFF
      make -j$(nproc)
      make install

      Step 1: Create the Test App

    Here's how to deploy your test_zmq application to the Trenz board 
      
      Create temp directory on trenz
        ssh petalinux@192.168.7.2
        sudo mkdir -p /temp
      Use scp to Transfer the Binary
        scp ./install/usr/bin/test_zmq petalinux@192.168.7.2:/temp/
      Locate the libzmq.so.5 file on your host machine:
        find ~/projects/rover/ -name "libzmq.so.5*"
      Copy the .so file to the Trenz board:
          scp /software/libzmq/sysroot-stage/usr/lib/libzmq.so.5* petalinux@192.168.7.2:/temp/
      On the Trenz board, move it to a location like /usr/lib or update the LD_LIBRARY_PATH:
        ssh petalinux@192.168.7.2
        sudo cp /temp/libzmq.so.5* /usr/lib/
        sudo ldconfig
      Run app
        ssh petalinux@192.168.7.2
        chmod +x /temp/test_zmq
        /temp/test_zmq
